struct Coordinate {
    double lat;
    double lon;
    
    Coordinate() : lat(0), lon(0) {}
    Coordinate(double lat, double lon) : lat(lat), lon(lon) {}
    
    bool operator==(const Coordinate& other) const {
        return std::abs(lat - other.lat) < 1e-9 && std::abs(lon - other.lon) < 1e-9;
    }
    
    size_t hash() const {
        return std::hash<double>{}(lat) ^ (std::hash<double>{}(lon) << 1);
    }
    
    double distanceTo(const Coordinate& other) const {
        double dLat = (other.lat - lat) * DEG_TO_RAD;
        double dLon = (other.lon - lon) * DEG_TO_RAD;
        double a = std::sin(dLat/2) * std::sin(dLat/2) +
                   std::cos(lat * DEG_TO_RAD) * std::cos(other.lat * DEG_TO_RAD) *
                   std::sin(dLon/2) * std::sin(dLon/2);
        double c = 2 * std::atan2(std::sqrt(a), std::sqrt(1-a));
        return EARTH_RADIUS * c;
    }
    
    Coordinate toMercator() const {
        double x = lon * DEG_TO_RAD * EARTH_RADIUS;
        double y = std::log(std::tan(PI/4 + lat * DEG_TO_RAD / 2)) * EARTH_RADIUS;
        return Coordinate(y, x);
    }
    
    static Coordinate fromMercator(double x, double y) {
        double lon = x / EARTH_RADIUS * RAD_TO_DEG;
        double lat = (2 * std::atan(std::exp(y / EARTH_RADIUS)) - PI/2) * RAD_TO_DEG;
        return Coordinate(lat, lon);
    }
};

// Bezier curve interpolation
Coordinate bezierPoint(const Coordinate& p0, const Coordinate& p1, 
                       const Coordinate& p2, const Coordinate& p3, double t) {
    double mt = 1 - t;
    double mt2 = mt * mt;
    double mt3 = mt2 * mt;
    double t2 = t * t;
    double t3 = t2 * t;
    
    return Coordinate(
        mt3 * p0.lat + 3 * mt2 * t * p1.lat + 3 * mt * t2 * p2.lat + t3 * p3.lat,
        mt3 * p0.lon + 3 * mt2 * t * p1.lon + 3 * mt * t2 * p2.lon + t3 * p3.lon
    );
}

// Point to segment projection
std::pair<double, Coordinate> projectToSegment(const Coordinate& point,
                                               const Coordinate& segStart,
                                               const Coordinate& segEnd) {
    double dx = segEnd.lon - segStart.lon;
    double dy = segEnd.lat - segStart.lat;
    double len2 = dx * dx + dy * dy;
    
    if (len2 == 0) return {point.distanceTo(segStart), segStart};
    
    double t = ((point.lon - segStart.lon) * dx + (point.lat - segStart.lat) * dy) / len2;
    t = clamp(t, 0.0, 1.0);
    
    Coordinate proj(
        segStart.lat + t * dy,
        segStart.lon + t * dx
    );
    
    return {point.distanceTo(proj), proj};
}

class SpatialGrid {
private:
    struct Cell {
        std::vector<int> stationIndices;
        std::vector<int> segmentIndices;
    };
    
    std::unordered_map<int, Cell> grid;
    double cellSize;
    int gridSize;
    
    int getCellIndex(const Coordinate& coord) const {
        int latIdx = static_cast<int>((coord.lat + 90) / cellSize);
        int lonIdx = static_cast<int>((coord.lon + 180) / cellSize);
        return latIdx * gridSize + lonIdx;
    }
    
public:
    SpatialGrid(double size = 0.01, int sizeInt = 36000)
        : cellSize(size), gridSize(sizeInt) {}
    
    void clear() {
        grid.clear();
    }
    
    void insertStation(int index, const Coordinate& coord) {
        grid[getCellIndex(coord)].stationIndices.push_back(index);
    }
    
    void insertSegment(int index, const Coordinate& start, const Coordinate& end) {
        grid[getCellIndex(start)].segmentIndices.push_back(index);
        grid[getCellIndex(end)].segmentIndices.push_back(index);
    }
    
    std::vector<int> queryNearbyStations(const Coordinate& coord, double radius) const {
        std::vector<int> results;
        int cellRadius = static_cast<int>(radius / cellSize) + 1;
        
        int centerIdx = getCellIndex(coord);
        int centerLat = centerIdx / gridSize;
        int centerLon = centerIdx % gridSize;
        
        for (int lat = centerLat - cellRadius; lat <= centerLat + cellRadius; ++lat) {
            for (int lon = centerLon - cellRadius; lon <= centerLon + cellRadius; ++lon) {
                if (lat < 0 || lat >= gridSize || lon < 0 || lon >= gridSize) continue;
                
                int idx = lat * gridSize + lon;
                auto it = grid.find(idx);
                if (it != grid.end()) {
                    results.insert(results.end(), it->second.stationIndices.begin(),
                                   it->second.stationIndices.end());
                }
            }
        }
        
        return results;
    }
    
    std::vector<int> queryNearbySegments(const Coordinate& coord, double radius) const {
        std::vector<int> results;
        int cellRadius = static_cast<int>(radius / cellSize) + 1;
        
        int centerIdx = getCellIndex(coord);
        int centerLat = centerIdx / gridSize;
        int centerLon = centerIdx % gridSize;
        
        for (int lat = centerLat - cellRadius; lat <= centerLat + cellRadius; ++lat) {
            for (int lon = centerLon - cellRadius; lon <= centerLon + cellRadius; ++lon) {
                if (lat < 0 || lat >= gridSize || lon < 0 || lon >= gridSize) continue;
                
                int idx = lat * gridSize + lon;
                auto it = grid.find(idx);
                if (it != grid.end()) {
                    results.insert(results.end(), it->second.segmentIndices.begin(),
                                   it->second.segmentIndices.end());
                }
            }
        }
        
        return results;
    }
};

class GeometryEngine {
private:
    geom::GeometryFactory::Ptr factory;
    mutable SpatialGrid spatialIndex;
    
public:
    GeometryEngine() : factory(geom::GeometryFactory::create()) {}
    
    // Point in polygon test using ray casting
    bool pointInPolygon(const Coordinate& point, const std::vector<Coordinate>& polygon) const {
        if (polygon.size() < 3) return false;
        
        bool inside = false;
        size_t n = polygon.size();
        
        for (size_t i = 0, j = n - 1; i < n; j = i++) {
            double xi = polygon[i].lon, yi = polygon[i].lat;
            double xj = polygon[j].lon, yj = polygon[j].lat;
            
            if (((yi > point.lat) != (yj > point.lat)) &&
                (point.lon < (xj - xi) * (point.lat - yi) / (yj - yi) + xi)) {
                inside = !inside;
            }
        }
        
        return inside;
    }
    
    // Find nearest point on polyline
    std::pair<double, Coordinate> snapToPolyline(const Coordinate& point,
                                                  const std::vector<Coordinate>& polyline) const {
        if (polyline.empty()) return {std::numeric_limits<double>::infinity(), point};
        
        double minDist = std::numeric_limits<double>::infinity();
        Coordinate nearest = point;
        
        for (size_t i = 0; i < polyline.size() - 1; ++i) {
            auto [dist, proj] = projectToSegment(point, polyline[i], polyline[i + 1]);
            if (dist < minDist) {
                minDist = dist;
                nearest = proj;
            }
        }
        
        return {minDist, nearest};
    }
    
    // Buffer around point (circle)
    std::vector<Coordinate> createCircle(const Coordinate& center, double radiusMeters, int segments = 32) const {
        std::vector<Coordinate> result;
        double radiusDeg = radiusMeters / EARTH_RADIUS * RAD_TO_DEG;
        
        for (int i = 0; i < segments; ++i) {
            double angle = 2 * PI * i / segments;
            result.emplace_back(
                center.lat + radiusDeg * std::sin(angle),
                center.lon + radiusDeg * std::cos(angle) / std::cos(center.lat * DEG_TO_RAD)
            );
        }
        
        return result;
    }
    
    // Merge stations within threshold
    std::vector<Station> mergeStations(const std::vector<Station>& stations, double threshold) const {
        std::vector<Station> merged;
        std::set<int> used;
        
        spatialIndex.clear();
        for (size_t i = 0; i < stations.size(); ++i) {
            spatialIndex.insertStation(i, stations[i].coord);
        }
        
        for (size_t i = 0; i < stations.size(); ++i) {
            if (used.count(i)) continue;
            
            Station hub = stations[i];
            hub.isInterchange = false;
            hub.lines = stations[i].lines;
            
            auto nearby = spatialIndex.queryNearbyStations(stations[i].coord, threshold);
            for (int idx : nearby) {
                if (used.count(idx)) continue;
                if (stations[idx].coord.distanceTo(stations[i].coord) < threshold) {
                    used.insert(idx);
                    hub.isInterchange = true;
                    for (const QString& line : stations[idx].lines) {
                        if (!hub.lines.contains(line)) hub.lines.append(line);
                    }
                }
            }
            
            merged.push_back(hub);
        }
        
        return merged;
    }
    
    // Simplify polyline using Douglas-Peucker
    std::vector<Coordinate> simplifyPolyline(const std::vector<Coordinate>& polyline, double epsilon) const {
        if (polyline.size() < 3) return polyline;
        
        double maxDist = 0;
        int maxIdx = 0;
        
        for (size_t i = 1; i < polyline.size() - 1; ++i) {
            auto [dist, _] = projectToSegment(polyline[i], polyline.front(), polyline.back());
            if (dist > maxDist) {
                maxDist = dist;
                maxIdx = i;
            }
        }
        
        if (maxDist > epsilon) {
            auto left = simplifyPolyline(std::vector<Coordinate>(polyline.begin(), polyline.begin() + maxIdx + 1), epsilon);
            auto right = simplifyPolyline(std::vector<Coordinate>(polyline.begin() + maxIdx, polyline.end()), epsilon);
            left.insert(left.end(), right.begin(), right.end());
            return left;
        }
        
        return {polyline.front(), polyline.back()};
    }
    
    SpatialGrid& getSpatialIndex() { return spatialIndex; }
    const SpatialGrid& getSpatialIndex() const { return spatialIndex; }
};

class RoutingGraph {
private:
    std::unordered_map<int, RoutingNode> nodes;
    std::unordered_map<int, std::vector<int>> adjacency;
    mutable SpatialGrid spatialIndex;
    
public:
    void clear() {
        nodes.clear();
        adjacency.clear();
        spatialIndex.clear();
    }
    
    void addNode(int id, const Coordinate& coord) {
        RoutingNode node;
        node.id = id;
        node.coord = coord;
        nodes[id] = node;
        spatialIndex.insertStation(id, coord);
    }
    
    void addEdge(int from, int to, double weight) {
        adjacency[from].push_back(to);
        nodes[from].neighbors.push_back(to);
        nodes[from].edgeWeights.push_back(weight);
    }
    
    std::vector<Coordinate> findPath(const Coordinate& start, const Coordinate& end) const {
        // Find nearest nodes
        auto startNearby = spatialIndex.queryNearbyStations(start, 1000);
        auto endNearby = spatialIndex.queryNearbyStations(end, 1000);
        
        if (startNearby.empty() || endNearby.empty()) return {};
        
        int startNode = startNearby[0];
        int endNode = endNearby[0];
        
        // A* algorithm
        std::unordered_map<int, double> gScore;
        std::unordered_map<int, double> fScore;
        std::unordered_map<int, int> cameFrom;
        std::set<std::pair<double, int>> openSet;
        
        gScore[startNode] = 0;
        fScore[startNode] = nodes.at(startNode).coord.distanceTo(end);
        openSet.insert({fScore[startNode], startNode});
        
        while (!openSet.empty()) {
            int current = openSet.begin()->second;
            
            if (current == endNode) {
                // Reconstruct path
                std::vector<Coordinate> path;
                path.push_back(nodes.at(endNode).coord);
                
                while (cameFrom.count(current)) {
                    current = cameFrom[current];
                    path.push_back(nodes.at(current).coord);
                }
                
                std::reverse(path.begin(), path.end());
                return path;
            }
            
            openSet.erase(openSet.begin());
            
            for (size_t i = 0; i < nodes.at(current).neighbors.size(); ++i) {
                int neighbor = nodes.at(current).neighbors[i];
                double weight = nodes.at(current).edgeWeights[i];
                double tentativeG = gScore[current] + weight;
                
                if (!gScore.count(neighbor) || tentativeG < gScore[neighbor]) {
                    cameFrom[neighbor] = current;
                    gScore[neighbor] = tentativeG;
                    fScore[neighbor] = tentativeG + nodes.at(neighbor).coord.distanceTo(end);
                    openSet.insert({fScore[neighbor], neighbor});
                }
            }
        }
        
        return {};
    }
    
    void buildFromTracks(const std::vector<RailwayTrack>& tracks) {
        clear();
        int nodeId = 0;
        
        for (const auto& track : tracks) {
            if (track.geometry.empty()) continue;
            
            int prevId = nodeId++;
            addNode(prevId, track.geometry[0]);
            
            for (size_t i = 1; i < track.geometry.size(); ++i) {
                int currId = nodeId++;
                addNode(currId, track.geometry[i]);
                
                double dist = track.geometry[i-1].distanceTo(track.geometry[i]);
                addEdge(prevId, currId, dist);
                addEdge(currId, prevId, dist);
                
                prevId = currId;
            }
        }
    }
};

    std::string responseBuffer;
    QMutex mutex;
    QThreadPool threadPool;
    
    static size_t writeCallback(char* ptr, size_t size, size_t nmemb, void* userdata) {
        NetworkManager* manager = static_cast<NetworkManager*>(userdata);
        size_t totalSize = size * nmemb;
        manager->responseBuffer.append(ptr, totalSize);
        return totalSize;
    }
    
public:
    NetworkManager() {
        curl_global_init(CURL_GLOBAL_ALL);
        curl = curl_easy_init();
        curl_easy_setopt(curl, CURLOPT_WRITEFUNCTION, writeCallback);
        curl_easy_setopt(curl, CURLOPT_WRITEDATA, this);
        curl_easy_setopt(curl, CURLOPT_TIMEOUT, 30L);
        curl_easy_setopt(curl, CURLOPT_FOLLOWLOCATION, 1L);
        curl_easy_setopt(curl, CURLOPT_USERAGENT, "LondonTransport/1.0");
    }
    
    ~NetworkManager() {
        if (curl) curl_easy_cleanup(curl);
        curl_global_cleanup();
    }
    
    std::future<std::optional<QString>> getAsync(const QString& url) {
        return std::async(std::launch::async, [this, url]() -> std::optional<QString> {
            QMutexLocker locker(&mutex);
            responseBuffer.clear();
            
            curl_easy_setopt(curl, CURLOPT_URL, url.toUtf8().constData());
            
            CURLcode res = curl_easy_perform(curl);
            if (res != CURLE_OK) {
                return std::nullopt;
            }
            
            return QString::fromStdString(responseBuffer);
        });
    }
    
    std::optional<QString> get(const QString& url) {
        QMutexLocker locker(&mutex);
        responseBuffer.clear();
        
        curl_easy_setopt(curl, CURLOPT_URL, url.toUtf8().constData());
        
        CURLcode res = curl_easy_perform(curl);
        if (res != CURLE_OK) {
            return std::nullopt;
        }
        
        return QString::fromStdString(responseBuffer);
    }
    
    std::future<std::optional<json>> getJsonAsync(const QString& url) {
        return std::async(std::launch::async, [this, url]() -> std::optional<json> {
            auto response = get(url);
            if (!response) return std::nullopt;
            
            try {
                return json::parse(response->toStdString());
            } catch (...) {
                return std::nullopt;
            }
        });
    }
    
    std::optional<json> getJson(const QString& url) {
        auto response = get(url);
        if (!response) return std::nullopt;
        
        try {
            return json::parse(response->toStdString());
        } catch (...) {
            return std::nullopt;
        }
    }
};

// CACHE / PERSISTENCE (SQLite)
// ============================================================================

class CacheManager {
private:
    sqlite3* db;
    QString dbPath;
    QMutex mutex;
    
    bool executeQuery(const QString& sql) {
        char* errMsg = nullptr;
        int result = sqlite3_exec(db, sql.toUtf8().constData(), nullptr, nullptr, &errMsg);
        if (result != SQLITE_OK) {
            qWarning() << "SQL error:" << errMsg;
            sqlite3_free(errMsg);
            return false;
        }
        return true;
    }
    
public:
    CacheManager() : db(nullptr) {
        QString cacheDir = QStandardPaths::writableLocation(QStandardPaths::CacheLocation);
        QDir().mkpath(cacheDir);
        dbPath = cacheDir + "/london_transport_cache.db";
        
        if (sqlite3_open(dbPath.toUtf8().constData(), &db) == SQLITE_OK) {
            initializeTables();
        }
    }
    
    ~CacheManager() {
        if (db) sqlite3_close(db);
    }
    
    void initializeTables() {
        executeQuery("CREATE TABLE IF NOT EXISTS api_cache ("
                     "key TEXT PRIMARY KEY, "
                     "data TEXT, "
                     "timestamp INTEGER, "
                     "expiry INTEGER)");
        
        executeQuery("CREATE TABLE IF NOT EXISTS custom_lines ("
                     "id TEXT PRIMARY KEY, "
                     "name TEXT, "
                     "color TEXT, "
                     "geometry TEXT, "
                     "stations TEXT)");
        
        executeQuery("CREATE TABLE IF NOT EXISTS free_stations ("
                     "id TEXT PRIMARY KEY, "
                     "name TEXT, "
                     "lat REAL, "
                     "lon REAL)");
    }
    
    void put(const QString& key, const QString& data, int expiryMs = CACHE_EXPIRY_DAYS * 24 * 3600 * 1000) {
        QMutexLocker locker(&mutex);
        qint64 now = QDateTime::currentMSecsSinceEpoch();
        qint64 expiry = now + expiryMs;
        
        QString sql = QString("INSERT OR REPLACE INTO api_cache (key, data, timestamp, expiry) "
                              "VALUES ('%1', '%2', %3, %4)")
                          .arg(key, QString(data).replace("'", "''")).arg(now).arg(expiry);
        executeQuery(sql);
    }
    
    std::optional<QString> get(const QString& key) {
        QMutexLocker locker(&mutex);
        qint64 now = QDateTime::currentMSecsSinceEpoch();
        
        QString sql = QString("SELECT data, expiry FROM api_cache WHERE key = '%1'").arg(key);
        
        sqlite3_stmt* stmt;
        if (sqlite3_prepare_v2(db, sql.toUtf8().constData(), -1, &stmt, nullptr) != SQLITE_OK) {
            return std::nullopt;
        }
        
        std::optional<QString> result;
        if (sqlite3_step(stmt) == SQLITE_ROW) {
            qint64 expiry = sqlite3_column_int64(stmt, 1);
            if (expiry > now) {
                result = QString::fromUtf8(reinterpret_cast<const char*>(sqlite3_column_text(stmt, 0)));
            } else {
                // Expired, delete it
                QString deleteSql = QString("DELETE FROM api_cache WHERE key = '%1'").arg(key);
                executeQuery(deleteSql);
            }
        }
        
        sqlite3_finalize(stmt);
        return result;
    }
    
    void saveCustomLine(const Line& line) {
        QString geometryJson;
        QJsonArray geometryArray;
        for (const auto& coord : line.geometry) {
            QJsonArray coordArray;
            coordArray.append(coord.lat);
            coordArray.append(coord.lon);
            geometryArray.append(coordArray);
        }
        QJsonDocument doc(geometryArray);
        geometryJson = doc.toJson(QJsonDocument::Compact);
        
        QString sql = QString("INSERT OR REPLACE INTO custom_lines (id, name, color, geometry, stations) "
                              "VALUES ('%1', '%2', '%3', '%4', '')")
                          .arg(line.id, line.name, line.color, geometryJson.replace("'", "''"));
        executeQuery(sql);
    }
    
    void saveFreeStation(const Station& station) {
        QString sql = QString("INSERT OR REPLACE INTO free_stations (id, name, lat, lon) "
                              "VALUES ('%1', '%2', %3, %4)")
                          .arg(station.id, station.name).arg(station.coord.lat).arg(station.coord.lon);
        executeQuery(sql);
    }
    
    std::vector<Line> loadCustomLines() {
        std::vector<Line> lines;
        QString sql = "SELECT id, name, color, geometry FROM custom_lines";
        
        sqlite3_stmt* stmt;
        if (sqlite3_prepare_v2(db, sql.toUtf8().constData(), -1, &stmt, nullptr) != SQLITE_OK) {
            return lines;
        }
        
        while (sqlite3_step(stmt) == SQLITE_ROW) {
            Line line;
            line.id = QString::fromUtf8(reinterpret_cast<const char*>(sqlite3_column_text(stmt, 0)));
            line.name = QString::fromUtf8(reinterpret_cast<const char*>(sqlite3_column_text(stmt, 1)));
            line.color = QString::fromUtf8(reinterpret_cast<const char*>(sqlite3_column_text(stmt, 2)));
            line.isCustom = true;
            
            QString geometryJson = QString::fromUtf8(reinterpret_cast<const char*>(sqlite3_column_text(stmt, 3)));
            QJsonDocument doc = QJsonDocument::fromJson(geometryJson.toUtf8());
            QJsonArray geometryArray = doc.array();
            
            for (const auto& coordValue : geometryArray) {
                QJsonArray coordArray = coordValue.toArray();
                line.geometry.emplace_back(coordArray[0].toDouble(), coordArray[1].toDouble());
            }
            
            lines.push_back(line);
        }
        
        sqlite3_finalize(stmt);
        return lines;
    }
    
    std::vector<Station> loadFreeStations() {
        std::vector<Station> stations;
        QString sql = "SELECT id, name, lat, lon FROM free_stations";
        
        sqlite3_stmt* stmt;
        if (sqlite3_prepare_v2(db, sql.toUtf8().constData(), -1, &stmt, nullptr) != SQLITE_OK) {
            return stations;
        }
        
        while (sqlite3_step(stmt) == SQLITE_ROW) {
            Station station;
            station.id = QString::fromUtf8(reinterpret_cast<const char*>(sqlite3_column_text(stmt, 0)));
            station.name = QString::fromUtf8(reinterpret_cast<const char*>(sqlite3_column_text(stmt, 1)));
            station.coord.lat = sqlite3_column_double(stmt, 2);
            station.coord.lon = sqlite3_column_double(stmt, 3);
            stations.push_back(station);
        }
        
        sqlite3_finalize(stmt);
        return stations;
    }
};

class MapWidget : public QWidget {

    QNetworkDiskCache* diskCache;
    QMap<QString, QPixmap> tileCache;
    QMutex renderMutex;
    
    // Tile server URLs
    static QString getTileUrl(const QString& theme, int x, int y, int zoom) {
        if (theme == "satellite") {
            return QString("https://server.arcgisonline.com/ArcGIS/rest/services/World_Imagery/MapServer/tile/%1/%2/%3").arg(zoom).arg(y).arg(x);
        } else if (theme == "dark") {
            return QString("https://a.basemaps.cartocdn.com/dark_all/%1/%2/%3.png").arg(zoom).arg(x).arg(y);
        } else if (theme == "voyager") {
            return QString("https://a.basemaps.cartocdn.com/rastertiles/voyager/%1/%2/%3.png").arg(zoom).arg(x).arg(y);
        } else {
            return QString("https://a.tile.openstreetmap.org/%1/%2/%3.png").arg(zoom).arg(x).arg(y);
        }
    }
    
    // Map projection helpers
    Coordinate screenToWorld(const QPoint& screen) const {
        double scale = TILE_SIZE * std::pow(2, zoom);
        double mercX = center.toMercator().lon + (screen.x() - width() / 2.0) / scale;
        double mercY = center.toMercator().lat - (screen.y() - height() / 2.0) / scale;
        return Coordinate::fromMercator(mercX, mercY);
    }
    
    QPoint worldToScreen(const Coordinate& world) const {
        double scale = TILE_SIZE * std::pow(2, zoom);
        Coordinate merc = world.toMercator();
        double dx = merc.lon - center.toMercator().lon;
        double dy = merc.lat - center.toMercator().lat;
        return QPoint(
            static_cast<int>(width() / 2.0 + dx * scale),
            static_cast<int>(height() / 2.0 - dy * scale)
        );
    }
    
public:

        : QWidget(parent), zoom(DEFAULT_ZOOM), center(51.5074, -0.1278),
          isDragging(false), mapTheme("light"), showHistorical(false),

protected:
    void paintEvent(QPaintEvent* event) override {
        Q_UNUSED(event);

        QPainter painter(this);
        painter.setRenderHint(QPainter::Antialiasing);
        
        // Draw background based on theme

        } else if (mapTheme == "satellite") {

            painter.setPen(QPen(QColor(0, 255, 100, 30), 1)); // Neon green radar grid
            for (int i = 0; i < width(); i += 50) painter.drawLine(i, 0, i, height());
            for (int i = 0; i < height(); i += 50) painter.drawLine(0, i, width(), i);
        } else {
            painter.fillRect(rect(), QColor(245, 245, 240));
        }
        
        // Draw map tiles (simplified - in production would fetch actual tiles)
        drawMapTiles(painter);
        
        // Draw railway tracks (historical/physical)

            drawTracks(painter);
        }

        if (showCanals) {

        }

        // Draw free stations

            drawTrains(painter);
        }

        if (showCoverage) {

    void drawMapTiles(QPainter& painter) {
        double scale = TILE_SIZE * std::pow(2, zoom);
        int tileX = static_cast<int>((center.toMercator().lon / (2 * PI) + 0.5) * std::pow(2, zoom));
        int tileY = static_cast<int>((1 - std::log(std::tan(center.lat * DEG_TO_RAD) + 1 / std::cos(center.lat * DEG_TO_RAD)) / PI) / 2 * std::pow(2, zoom));
        
        int startTileX = tileX - width() / (2 * scale) - 1;
        int startTileY = tileY - height() / (2 * scale) - 1;
        int endTileX = tileX + width() / (2 * scale) + 1;
        int endTileY = tileY + height() / (2 * scale) + 1;
        
        for (int tx = startTileX; tx <= endTileX; ++tx) {
            for (int ty = startTileY; ty <= endTileY; ++ty) {
                QString tileKey = QString("%1_%2_%3_%4").arg(mapTheme).arg(zoom).arg(tx).arg(ty);
                
                // Check if tile is in cache
                if (!tileCache.contains(tileKey)) {
                    // Fetch tile asynchronously
                    QString urlString = getTileUrl(mapTheme, tx, ty, static_cast<int>(zoom));
                    QUrl url(urlString);
                    QNetworkRequest request(url);
                    request.setAttribute(QNetworkRequest::CacheLoadControlAttribute, QNetworkRequest::PreferCache);
                    
                    QNetworkReply* reply = networkManager->get(request);
                    connect(reply, &QNetworkReply::finished, [this, reply, tileKey]() {
                        if (reply->error() == QNetworkReply::NoError) {
                            QPixmap pixmap;
                            if (pixmap.loadFromData(reply->readAll())) {
                                tileCache[tileKey] = pixmap;
                                update();
                            }
                        }
                        reply->deleteLater();
                    });
                    
                    // Draw placeholder while loading
                    painter.fillRect(
                        static_cast<int>((tx - tileX) * scale + width() / 2),
                        static_cast<int>((ty - tileY) * scale + height() / 2),
                        static_cast<int>(scale),
                        static_cast<int>(scale),
                        QColor(220, 220, 220)
                    );
                } else {
                    // Draw cached tile
                    QPixmap tile = tileCache.value(tileKey);
                    painter.drawPixmap(
                        static_cast<int>((tx - tileX) * scale + width() / 2),
                        static_cast<int>((ty - tileY) * scale + height() / 2),
                        static_cast<int>(scale),
                        static_cast<int>(scale),
                        tile
                    );
                }
            }
        }
    }
    
    void drawLines(QPainter& painter) {
        for (const auto& line : lines) {
            if (line.geometry.empty()) continue;
            
            QColor color = hexToColor(line.color);
            
            // Draw casing (black outline)
            QPen casingPen(QColor(0, 0, 0, 100), 8, Qt::SolidLine, Qt::RoundCap, Qt::RoundJoin);
            painter.setPen(casingPen);
            
            QPainterPath casingPath;
            casingPath.moveTo(worldToScreen(line.geometry[0]));
            for (size_t i = 1; i < line.geometry.size(); ++i) {
                casingPath.lineTo(worldToScreen(line.geometry[i]));
            }
            painter.drawPath(casingPath);
            
            // Draw core (brand color)
            QPen corePen(color, 4, Qt::SolidLine, Qt::RoundCap, Qt::RoundJoin);
            painter.setPen(corePen);
            
            QPainterPath corePath;
            corePath.moveTo(worldToScreen(line.geometry[0]));
            for (size_t i = 1; i < line.geometry.size(); ++i) {
                corePath.lineTo(worldToScreen(line.geometry[i]));
            }
            painter.drawPath(corePath);
        }
    }
    
    void drawStations(QPainter& painter) {
        if (zoom < 13) return;
        
        for (const auto& station : stations) {
            QPoint screenPos = worldToScreen(station.coord);
            
            // Skip if outside visible area
            if (!rect().contains(screenPos, true)) continue;
            
            // Draw station marker (roundel-style)
            QColor markerColor = station.isInterchange ? QColor(255, 255, 255) : QColor(0, 0, 0);
            painter.setBrush(QBrush(markerColor));
            painter.setPen(QPen(QColor(0, 0, 0), 1));
            painter.drawEllipse(screenPos, 6, 6);
            
            // Draw station label
            if (zoom >= 14) {
                painter.setPen(QPen(QColor(50, 50, 50), 1));
                painter.setFont(QFont("Arial", 8));
                painter.drawText(screenPos + QPoint(10, 5), station.name);
            }
        }
    }
    
    void drawFreeStations(QPainter& painter) {
        for (const auto& station : freeStations) {
            QPoint screenPos = worldToScreen(station.coord);
            
            // Draw catchment circle
            double scale = TILE_SIZE * std::pow(2, zoom);
            double radiusPixels = (CATCHMENT_RADIUS / EARTH_RADIUS * RAD_TO_DEG) * scale;
            
            painter.setBrush(QBrush(QColor(0, 150, 255, 50)));
            painter.setPen(QPen(QColor(0, 150, 255, 150), 1, Qt::DashLine));
            painter.drawEllipse(screenPos, static_cast<int>(radiusPixels), static_cast<int>(radiusPixels));
            
            // Draw station marker
            painter.setBrush(QBrush(QColor(0, 150, 255)));
            painter.setPen(QPen(QColor(0, 0, 0), 1));
            painter.drawEllipse(screenPos, 8, 8);
            
            // Draw label
            painter.setPen(QPen(QColor(50, 50, 50), 1));
            painter.setFont(QFont("Arial", 8));
            painter.drawText(screenPos + QPoint(12, 5), station.name);
        }
    }
    
    void drawTracks(QPainter& painter) {
        for (const auto& track : tracks) {
            if (track.geometry.empty()) continue;
            
            QColor color = track.isAbandoned ? QColor(150, 150, 150, 150) : QColor(100, 100, 100, 100);
            QPen pen(color, track.isAbandoned ? 2 : 3, track.isAbandoned ? Qt::DashLine : Qt::SolidLine);
            painter.setPen(pen);

    void mousePressEvent(QMouseEvent* event) override {
        if (event->button() == Qt::LeftButton) {
            isDragging = true;
            dragStart = event->pos();
            dragStartCenter = center;
            setCursor(Qt::ClosedHandCursor);
        }
        
        mouseScreenPos = event->pos();
        mouseWorldPos = screenToWorld(event->pos());
        
        emit mapClicked(mouseWorldPos, event->button());
        update();
    }
    
    void mouseMoveEvent(QMouseEvent* event) override {
        mouseScreenPos = event->pos();
        mouseWorldPos = screenToWorld(event->pos());
        
        if (isDragging) {
            double scale = TILE_SIZE * std::pow(2, zoom);
            double dx = (dragStart.x() - event->x()) / scale;
            double dy = (event->y() - dragStart.y()) / scale;
            
            center = Coordinate(
                dragStartCenter.lat + dy,
                dragStartCenter.lon - dx
            );
        }
        
        emit mouseMoved(mouseWorldPos);
        update();
    }
    
    void mouseReleaseEvent(QMouseEvent* event) override {
        if (event->button() == Qt::LeftButton) {
            isDragging = false;
            setCursor(Qt::OpenHandCursor);
        }
    }
    
    void wheelEvent(QWheelEvent* event) override {
        double delta = event->angleDelta().y() / 120.0;
        double newZoom = clamp(zoom + delta, MIN_ZOOM, MAX_ZOOM);
        
        // Zoom toward mouse position
        Coordinate mouseWorld = screenToWorld(event->position().toPoint());
        zoom = newZoom;
        Coordinate mouseWorldAfter = screenToWorld(event->position().toPoint());
        
        center.lat += mouseWorld.lat - mouseWorldAfter.lat;
        center.lon += mouseWorld.lon - mouseWorldAfter.lon;
        
        emit zoomChanged(zoom);
        update();
    }
    
    void keyPressEvent(QKeyEvent* event) override {
        if (event->key() == Qt::Key_Z && (event->modifiers() & Qt::ControlModifier)) {
            emit undoRequested();
        } else if (event->key() == Qt::Key_Y && (event->modifiers() & Qt::ControlModifier)) {
            emit redoRequested();
        } else if (event->key() == Qt::Key_Escape) {
            emit cancelRequested();
        }
    }
    
signals:
    void mapClicked(const Coordinate& coord, Qt::MouseButton button);
    void mouseMoved(const Coordinate& coord);
    void zoomChanged(double zoom);
    void undoRequested();
    void redoRequested();
    void cancelRequested();
};

    void parseLineData(const QString& lineId, const QString& jsonData) {
        try {
            json data = json::parse(jsonData.toStdString());
            
            Line line;
            line.id = lineId;
            line.name = TFL_LINE_NAMES.value(lineId, lineId);
            line.color = TFL_LINE_COLORS.value(lineId, "#888888");
            line.isCustom = false;
            
            // Parse route sequence
            if (data.contains("lineStrings")) {
                for (const auto& lineString : data["lineStrings"]) {
                    for (const auto& point : lineString) {
                        if (point.size() >= 2) {
                            line.geometry.emplace_back(point[1], point[0]);
                        }
                    }
                }
            }
            
            lines.push_back(line);
            
        } catch (...) {
            log("error", "Failed to parse line data for " + lineId);
        }
    }

    void log(const QString& level, const QString& message) {
        LogEntry entry(level, message);
        logs.push_back(entry);
        
        if (logs.size() > MAX_LOG_ENTRIES) {
            logs.erase(logs.begin());
        }
        
        // Add to log list
        QListWidgetItem* item = new QListWidgetItem();
        QString timestamp = entry.timestamp.toString("hh:mm:ss");
        item->setText(QString("[%1] %2: %3").arg(timestamp, level.toUpper(), message));
        
        if (level == "error") item->setForeground(QBrush(QColor(255, 100, 100)));
        else if (level == "warn") item->setForeground(QBrush(QColor(255, 200, 100)));
        else if (level == "info") item->setForeground(QBrush(QColor(100, 200, 255)));
        
        logList->addItem(item);
        logList->scrollToBottom();
        
        qDebug() << "[" << level << "]" << message;
    }

    // Map interactions
    void onMapClicked(const Coordinate& coord, Qt::MouseButton button) {
        if (button == Qt::RightButton) {
            showContextMenu(coord);
            return;
        }
        
        if (isDrawMode) {
            constructionState.currentDrawing.push_back(coord);
            
            if (constructionState.isAutoRoute) {
                // Auto-route using graph
                if (constructionState.currentDrawing.size() >= 2) {
                    auto last = constructionState.currentDrawing[constructionState.currentDrawing.size() - 2];
                    auto path = mapWidget->getRoutingGraph().findPath(last, coord);
                    if (!path.empty()) {
                        // Replace last segment with routed path
                        constructionState.currentDrawing.pop_back();
                        constructionState.currentDrawing.insert(
                            constructionState.currentDrawing.end(),
                            path.begin(), path.end()
                        );
                    }
                }
            }
            
            mapWidget->update();
        } else if (isBuildMode && !isDrawMode) {
            // Place free station
            Station station;
            station.id = generateId();
            station.name = "Station " + (freeStations.size() + 1);
            station.coord = coord;
            freeStations.push_back(station);
            cache.saveFreeStation(station);
            mapWidget->setFreeStations(freeStations);
            
            saveUndoState();
        }
    }

    // Undo/Redo
    void saveUndoState() {
        undoStack.push_back(constructionState);
        if (undoStack.size() > MAX_UNDO_STATES) {
            undoStack.erase(undoStack.begin());
        }
        redoStack.clear();
    }
    
    void undo() {
        if (undoStack.empty()) return;
        
        redoStack.push_back(constructionState);
        constructionState = undoStack.back();
        undoStack.pop_back();
        
        restoreConstructionState();
        log("info", "Undo performed");
    }
    
    void redo() {
        if (redoStack.empty()) return;
        
        undoStack.push_back(constructionState);
        constructionState = redoStack.back();
        redoStack.pop_back();
        
        restoreConstructionState();
        log("info", "Redo performed");
    }
